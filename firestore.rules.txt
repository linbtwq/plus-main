rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function owner(data) {
      return request.auth != null && data.userId == request.auth.uid;
    }
    function stat(value, max, duo) {
      // Парные оценки храним в прежнем формате, проверяя обе половины.
      return value is string && (duo
        ? (max == 10
          ? value.matches('([0-9]\\.[0-9]|10\\.0)/([0-9]\\.[0-9]|10\\.0)')
          : value.matches('([0-4]\\.[0-9]|5\\.0)/([0-4]\\.[0-9]|5\\.0)'))
        : (max == 10 ? value.matches('[0-9]\\.[0-9]|10\\.0') : value.matches('[0-4]\\.[0-9]|5\\.0')));
    }
    function validMetadata(m) {
      return m is map && m.keys().hasAll(['year','runtime','directors','creators','actors','countries'])
        && m.keys().hasOnly(['year','runtime','directors','creators','actors','countries'])
        && m.year is int && (m.year == 0 || (m.year >= 1870 && m.year <= 2200))
        && m.runtime is number && m.runtime >= 0 && m.runtime <= 10000
        && m.directors is list && m.directors.size() <= 10 && m.creators is list && m.creators.size() <= 10
        && m.actors is list && m.actors.size() <= 20 && m.countries is list && m.countries.size() <= 30;
    }
    function validMovie(d) {
      return d.keys().hasAll(['userId', 'title', 'score', 'type', 'poster', 'genres', 'authorName', 'authorPhoto', 'timestamp', 'date', 'stats', 'review', 'isDuo', 'tmdbId', 'mediaType', 'schemaVersion'])
        && d.keys().hasOnly(['userId', 'title', 'score', 'type', 'poster', 'genres', 'authorName', 'authorPhoto', 'timestamp', 'date', 'stats', 'review', 'isDuo', 'tmdbId', 'mediaType', 'schemaVersion', 'metadata', 'ratingMode', 'jointPartnerId', 'jointPartnerName', 'jointFirstId'])
        && (!('metadata' in d) || validMetadata(d.metadata))
        && d.get('ratingMode', 'detailed') in ['quick', 'detailed']
        && d.get('jointPartnerId', '') is string && d.get('jointPartnerId', '').size() <= 128
        && d.get('jointFirstId', '') is string && d.get('jointFirstId', '').size() <= 128
        && d.get('jointPartnerName', '') is string && d.get('jointPartnerName', '').size() <= 200
        && d.userId is string && d.schemaVersion == 2
        && d.title is string && d.title.size() > 0 && d.title.size() <= 500
        && d.score is number && d.score >= 0 && d.score <= 10
        && d.type in ['movie', 'series', 'anime'] && d.mediaType in ['movie', 'tv']
        && d.tmdbId is int && d.tmdbId > 0 && d.tmdbId <= 9999999999
        && d.poster is string && (d.poster == '' || d.poster.matches('https://image[.]tmdb[.]org/t/p/w500/[A-Za-z0-9._-]+'))
        && d.genres is string && d.genres.size() <= 500
        && d.authorName is string && d.authorName.size() <= 200
        && (d.authorPhoto == null || (d.authorPhoto is string && d.authorPhoto.size() <= 2000))
        && d.timestamp is timestamp && d.date is string && d.date.size() <= 40
        && (d.review == null || (d.review is string && d.review.size() <= 10000))
        && d.isDuo is bool && d.stats is map
        && d.stats.keys().hasAll(['script', 'actors', 'directing', 'chars', 'idea', 'atmosphere', 'impression'])
        && d.stats.keys().hasOnly(['script', 'actors', 'directing', 'chars', 'idea', 'atmosphere', 'impression'])
        && stat(d.stats.script, 10, d.isDuo) && stat(d.stats.actors, 10, d.isDuo)
        && stat(d.stats.directing, 10, d.isDuo) && stat(d.stats.chars, 10, d.isDuo)
        && stat(d.stats.idea, 10, d.isDuo) && stat(d.stats.atmosphere, 5, d.isDuo)
        && stat(d.stats.impression, 5, d.isDuo);
    }
    // Подтверждённая дружба разрешает чтение оценок, но не их изменение.
    function confirmedFriend(userId) {
      let direct = /databases/$(database)/documents/friendships/$(request.auth.uid + ':' + userId);
      let reverse = /databases/$(database)/documents/friendships/$(userId + ':' + request.auth.uid);
      return (exists(direct) && get(direct).data.status == 'accepted')
        || (exists(reverse) && get(reverse).data.status == 'accepted');
    }
    match /movies/{movieId} {
      // Транзакция читает отсутствие только новой личной записи или своей копии приглашения.
      allow get: if request.auth != null && !exists(/databases/$(database)/documents/movies/$(movieId))
        && (movieId.matches('joint-source-' + request.auth.uid + '-[a-f0-9-]{36}')
          || movieId.matches('joint-[A-Za-z0-9_-]{1,128}:' + request.auth.uid + ':[A-Za-z0-9_-]{1,500}'));

      allow read: if request.auth != null && (owner(resource.data) || confirmedFriend(resource.data.userId));
      allow delete: if owner(resource.data);
      allow create: if owner(request.resource.data) && validMovie(request.resource.data)
        && request.resource.data.timestamp == request.time;
      allow update: if owner(resource.data) && owner(request.resource.data)
        && request.resource.data.userId == resource.data.userId
        && validMovie(request.resource.data)
        && request.resource.data.timestamp == resource.data.timestamp;
    }
    // Совместная оценка отправляется как снимок; получатель сам подтверждает запись в своём архиве.
    match /jointRatings/{jointId} {
      function member(d) { return request.auth != null && request.auth.uid in d.members; }
      function validJoint(d) {
        return d.keys().hasAll(['senderId','receiverId','members','sourceMovieId','movie','status','createdAt','updatedAt'])
          && d.keys().hasOnly(['senderId','receiverId','members','sourceMovieId','movie','status','createdAt','updatedAt'])
          && d.senderId is string && d.receiverId is string
          && d.senderId.matches('[A-Za-z0-9_-]{1,128}') && d.receiverId.matches('[A-Za-z0-9_-]{1,128}')
          && d.senderId != d.receiverId && d.members == [d.senderId,d.receiverId]
          && d.sourceMovieId is string && d.sourceMovieId.matches('[A-Za-z0-9_-]{1,500}')
          && jointId == d.senderId + ':' + d.receiverId + ':' + d.sourceMovieId
          && d.status in ['pending','accepted','declined','cancelled']
          && d.movie is map && d.movie.userId == d.senderId && d.movie.isDuo == true
          && d.movie.get('jointPartnerId', '') == d.receiverId && d.movie.get('jointFirstId', '') == d.senderId
          && d.createdAt is timestamp && d.updatedAt is timestamp;
      }
      // Формат исходника и копии проверяется правилами movies, содержимое связывается точным сравнением.
      function sameSource(d) { return getAfter(/databases/$(database)/documents/movies/$(d.sourceMovieId)).data == d.movie; }
      function acceptedCopy(d) {
        let copy = getAfter(/databases/$(database)/documents/movies/$('joint-' + jointId)).data;
        return copy.userId == d.receiverId && copy.get('jointPartnerId', '') == d.senderId
          && copy.diff(d.movie).affectedKeys().hasOnly(['userId','authorName','authorPhoto','timestamp','jointPartnerId','jointPartnerName']);
      }
      allow get: if request.auth != null && !exists(/databases/$(database)/documents/jointRatings/$(jointId))
        && jointId.matches(request.auth.uid + ':[A-Za-z0-9_-]{1,128}:[A-Za-z0-9_-]{1,500}');
      allow read: if member(resource.data);
      allow create: if request.auth != null && request.auth.uid == request.resource.data.senderId
        && validJoint(request.resource.data) && request.resource.data.status == 'pending'
        && confirmedFriend(request.resource.data.receiverId) && sameSource(request.resource.data)
        && request.resource.data.createdAt == request.time && request.resource.data.updatedAt == request.time;
      allow update: if member(resource.data) && validJoint(request.resource.data)
        && request.resource.data.createdAt == resource.data.createdAt
        && request.resource.data.updatedAt == request.time
        && ((request.auth.uid == resource.data.senderId
          && resource.data.status in ['pending','declined','cancelled']
          && ((request.resource.data.status == 'pending' && confirmedFriend(resource.data.receiverId)
            && sameSource(request.resource.data)
            && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['status','movie','updatedAt']))
            || (resource.data.status == 'pending' && request.resource.data.status == 'cancelled'
              && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['status','updatedAt']))))
          || (request.auth.uid == resource.data.receiverId && resource.data.status == 'pending'
            && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['status','updatedAt'])
            && (request.resource.data.status == 'declined'
              || (request.resource.data.status == 'accepted' && confirmedFriend(resource.data.senderId) && acceptedCopy(request.resource.data)))));
    }
    // Планы видны владельцу и подтверждённым друзьям для совместного выбора.
    match /plans/{planId} {
      allow get: if request.auth != null && planId.matches(request.auth.uid + ':(movie|tv):[0-9]+');
      allow read: if request.auth != null && (owner(resource.data) || confirmedFriend(resource.data.userId));
      allow delete: if owner(resource.data);
      allow create, update: if owner(request.resource.data)
        && planId == request.auth.uid + ':' + request.resource.data.mediaType + ':' + string(request.resource.data.tmdbId)
        && request.resource.data.keys().hasAll(['userId','tmdbId','mediaType','title','poster','genres','year','createdAt'])
        && request.resource.data.keys().hasOnly(['userId','tmdbId','mediaType','title','poster','genres','year','createdAt'])
        && request.resource.data.tmdbId is int && request.resource.data.tmdbId > 0
        && request.resource.data.mediaType in ['movie','tv']
        && request.resource.data.title is string && request.resource.data.title.size() > 0 && request.resource.data.title.size() <= 500
        && request.resource.data.poster is string && (request.resource.data.poster == '' || request.resource.data.poster.matches('https://image[.]tmdb[.]org/t/p/w500/[A-Za-z0-9._-]+'))
        && request.resource.data.genres is string && request.resource.data.genres.size() <= 500
        && request.resource.data.year is string && request.resource.data.year.size() <= 4
        && request.resource.data.createdAt is number;
    }
    match /roulette/{userId} {
      allow read, delete: if request.auth != null && request.auth.uid == userId;
      allow create, update: if request.auth != null && request.auth.uid == userId
        && request.resource.data.keys().hasOnly(['text'])
        && request.resource.data.text is string && request.resource.data.text.size() <= 100000;
    }
    // Профили содержат только общедоступные для участников сайта сведения, без почты.
    match /profiles/{userId} {
      function validProfileImage(value, maxSize) {
        return value is string && value.size() <= maxSize
          && (value == '' || (value.size() <= 2000 && value.matches('https://[^ ]+'))
            || value.matches('data:image/(webp|jpeg|png);base64,[A-Za-z0-9+/]+={0,2}'));
      }
      function validProfile(d) {
        return d.keys().hasAll(['displayName', 'bio', 'photoURL', 'createdAt', 'updatedAt'])
          && d.keys().hasOnly(['displayName', 'bio', 'photoURL', 'bannerURL', 'createdAt', 'updatedAt'])
          && d.displayName is string && d.displayName.size() > 0 && d.displayName.size() <= 60
          && d.bio is string && d.bio.size() <= 500
          && validProfileImage(d.photoURL, 100000)
          && validProfileImage(d.get('bannerURL', ''), 350000)
          && d.createdAt is timestamp && d.updatedAt is timestamp;
      }
      allow get: if request.auth != null;
      allow list: if false;
      allow create: if request.auth != null && request.auth.uid == userId
        && validProfile(request.resource.data)
        && request.resource.data.createdAt == request.time && request.resource.data.updatedAt == request.time;
      allow update: if request.auth != null && request.auth.uid == userId
        && validProfile(request.resource.data)
        && request.resource.data.createdAt == resource.data.createdAt
        && request.resource.data.updatedAt == request.time;
    }
    match /friendships/{friendshipId} {
      function member(d) { return request.auth != null && request.auth.uid in d.members; }
      function validFriendship(d) {
        return d.keys().hasAll(['members', 'senderId', 'status', 'createdAt', 'updatedAt'])
          && d.keys().hasOnly(['members', 'senderId', 'status', 'createdAt', 'updatedAt'])
          && d.members is list && d.members.size() == 2
          && d.members[0] is string && d.members[1] is string
          && d.members[0].matches('[A-Za-z0-9_-]{1,128}') && d.members[1].matches('[A-Za-z0-9_-]{1,128}')
          && d.members[0] != d.members[1]
          && friendshipId == d.members[0] + ':' + d.members[1]
          && d.senderId in d.members && d.status in ['pending', 'accepted']
          && d.createdAt is timestamp && d.updatedAt is timestamp;
      }
      // Чтение отсутствующей пары нужно для безопасного создания заявки транзакцией.
      allow get: if (resource != null && member(resource.data)) || (request.auth != null
        && !exists(/databases/$(database)/documents/friendships/$(friendshipId))
        && (friendshipId.matches(request.auth.uid + ':[A-Za-z0-9_-]{1,128}')
          || friendshipId.matches('[A-Za-z0-9_-]{1,128}:' + request.auth.uid)));
      allow list: if member(resource.data);
      allow create: if request.auth != null && validFriendship(request.resource.data)
        && request.resource.data.senderId == request.auth.uid && request.resource.data.status == 'pending'
        && request.resource.data.createdAt == request.time && request.resource.data.updatedAt == request.time
        && exists(/databases/$(database)/documents/profiles/$(request.resource.data.members[0]))
        && exists(/databases/$(database)/documents/profiles/$(request.resource.data.members[1]))
        && !exists(/databases/$(database)/documents/friendships/$(request.resource.data.members[1] + ':' + request.resource.data.members[0]));
      // Принять заявку может только получатель. Отправитель не подтверждает дружбу сам.
      allow update: if member(resource.data) && request.auth.uid != resource.data.senderId
        && resource.data.status == 'pending' && request.resource.data.status == 'accepted'
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['status', 'updatedAt'])
        && request.resource.data.updatedAt == request.time;
      allow delete: if member(resource.data);
    }
    // Журналы личные: друзья не видят входы, удалённые записи и историю изменений.
    function ownJournal(id, d) {
      return request.auth != null && d.userId == request.auth.uid && id.matches(request.auth.uid + ':[A-Za-z0-9:_-]{1,500}');
    }
    match /activityStats/{uid} {
      function validStats(d) {
        return d.keys().hasAll(['userId','watchedSeconds','completions','rouletteStarts','logins','activeDays','currentStreak','bestStreak','lastDay','updatedAt'])
          && d.keys().hasOnly(['userId','watchedSeconds','completions','rouletteStarts','logins','activeDays','currentStreak','bestStreak','lastDay','updatedAt'])
          && d.userId == uid && d.watchedSeconds is number && d.watchedSeconds >= 0 && d.watchedSeconds <= 1000000000000
          && d.completions is int && d.completions >= 0 && d.rouletteStarts is int && d.rouletteStarts >= 0
          && d.logins is int && d.logins >= 0 && d.activeDays is int && d.activeDays >= 0
          && d.currentStreak is int && d.currentStreak >= 0 && d.bestStreak is int && d.bestStreak >= d.currentStreak
          && d.lastDay is string && (d.lastDay == '' || d.lastDay.matches('[0-9]{4}-[0-9]{2}-[0-9]{2}'))
          && d.updatedAt == request.time;
      }
      allow get: if request.auth != null && request.auth.uid == uid;
      allow create, update: if request.auth != null && request.auth.uid == uid && validStats(request.resource.data);
    }
    match /activityDays/{id} {
      allow get: if request.auth != null && id.matches(request.auth.uid + ':[0-9]{4}-[0-9]{2}-[0-9]{2}');
      allow list: if owner(resource.data);
      allow create: if ownJournal(id,request.resource.data)
        && request.resource.data.keys().hasAll(['userId','date','timezone','createdAt'])
        && request.resource.data.keys().hasOnly(['userId','date','timezone','createdAt'])
        && id == request.auth.uid + ':' + request.resource.data.date
        && request.resource.data.date.matches('[0-9]{4}-[0-9]{2}-[0-9]{2}')
        && request.resource.data.timezone is string && request.resource.data.timezone.size() <= 100
        && request.resource.data.createdAt == request.time;
    }
    match /activityEvents/{id} {
      allow get: if request.auth != null && id.matches(request.auth.uid + ':[A-Za-z0-9:_-]{1,500}');
      allow list: if owner(resource.data);
      allow create: if ownJournal(id,request.resource.data)
        && request.resource.data.keys().hasAll(['userId','type','label','occurredAt','createdAt'])
        && request.resource.data.keys().hasOnly(['userId','type','label','occurredAt','createdAt'])
        && request.resource.data.type in ['login','roulette'] && request.resource.data.label is string && request.resource.data.label.size() <= 200
        && request.resource.data.occurredAt is int && request.resource.data.occurredAt > 0
        && request.resource.data.createdAt == request.time;
    }
    match /watchSessions/{id} {
      function validSession(d) {
        return ownJournal(id,d) && d.keys().hasAll(['userId','tmdbId','mediaType','title','source','season','episode','animeRelease','position','duration','watchedSeconds','completed','completionMethod','createdAt','updatedAt'])
          && d.keys().hasOnly(['userId','tmdbId','mediaType','title','source','season','episode','animeRelease','position','duration','watchedSeconds','completed','completionMethod','createdAt','updatedAt'])
          && d.tmdbId is int && d.tmdbId > 0 && d.mediaType in ['movie','tv']
          && d.title is string && d.title.size() > 0 && d.title.size() <= 500
          && d.source in ['collaps','vidsrc','2embed','superembed','vidlink','aniliberty']
          && d.season is int && d.season > 0 && d.episode is int && d.episode > 0 && d.animeRelease is int && d.animeRelease >= 0
          && d.position is number && d.position >= 0 && d.position <= 604800
          && d.duration is number && d.duration >= 0 && d.duration <= 604800
          && d.watchedSeconds is int && d.watchedSeconds >= 0 && d.watchedSeconds <= 604800
          && d.completed is bool && d.completionMethod in ['none','automatic','manual']
          && (d.completed ? d.completionMethod != 'none' : d.completionMethod == 'none')
          && d.createdAt is timestamp && d.updatedAt == request.time;
      }
      allow get: if request.auth != null && id.matches(request.auth.uid + ':[A-Za-z0-9:_-]{1,500}');
      allow list: if owner(resource.data);
      allow create: if validSession(request.resource.data) && request.resource.data.createdAt == request.time;
      allow update: if owner(resource.data) && validSession(request.resource.data)
        && request.resource.data.createdAt == resource.data.createdAt
        && request.resource.data.tmdbId == resource.data.tmdbId && request.resource.data.mediaType == resource.data.mediaType
        && request.resource.data.source == resource.data.source && request.resource.data.season == resource.data.season
        && request.resource.data.episode == resource.data.episode && request.resource.data.animeRelease == resource.data.animeRelease
        && request.resource.data.watchedSeconds >= resource.data.watchedSeconds
        && (!resource.data.completed || (request.resource.data.completed && request.resource.data.completionMethod == resource.data.completionMethod));
    }
    match /ratingHistory/{id} {
      allow read: if owner(resource.data);
      allow create: if ownJournal(id,request.resource.data)
        && request.resource.data.keys().hasAll(['userId','movieId','before','after','reason','createdAt'])
        && request.resource.data.keys().hasOnly(['userId','movieId','before','after','reason','createdAt'])
        && request.resource.data.movieId is string && request.resource.data.movieId.matches('[A-Za-z0-9:_-]{1,1000}')
        && request.resource.data.reason in ['rating','joint','metadata'] && request.resource.data.createdAt == request.time
        && request.resource.data.after.userId == request.auth.uid
        && request.resource.data.after == getAfter(/databases/$(database)/documents/movies/$(request.resource.data.movieId)).data
        && (request.resource.data.before == null
          ? !exists(/databases/$(database)/documents/movies/$(request.resource.data.movieId))
          : request.resource.data.before == get(/databases/$(database)/documents/movies/$(request.resource.data.movieId)).data);
    }
    match /deletedItems/{id} {
      allow read: if owner(resource.data);
      allow create: if request.auth != null
        && request.resource.data.keys().hasAll(['userId','actorId','collectionName','itemId','snapshot','createdAt'])
        && request.resource.data.keys().hasOnly(['userId','actorId','collectionName','itemId','snapshot','createdAt'])
        && request.resource.data.userId is string && id.matches(request.resource.data.userId + ':[A-Za-z0-9:_-]{1,500}')
        && request.resource.data.actorId == request.auth.uid
        && request.resource.data.collectionName in ['movies','plans','friendships']
        && request.resource.data.itemId is string && request.resource.data.itemId.matches('[A-Za-z0-9:_-]{1,1000}')
        && request.resource.data.snapshot == get(/databases/$(database)/documents/$(request.resource.data.collectionName)/$(request.resource.data.itemId)).data
        && !existsAfter(/databases/$(database)/documents/$(request.resource.data.collectionName)/$(request.resource.data.itemId))
        && (request.resource.data.collectionName == 'friendships'
          ? (request.auth.uid in request.resource.data.snapshot.members && request.resource.data.userId in request.resource.data.snapshot.members)
          : (request.resource.data.userId == request.auth.uid && request.resource.data.snapshot.userId == request.auth.uid))
        && request.resource.data.createdAt == request.time;
    }
    // Награды постоянны; опыт и названия определяются каталогом, а не полями документа.
    match /achievements/{id} {
      allow get: if request.auth != null && id.matches(request.auth.uid + ':[a-z0-9-]{1,80}');
      allow read: if request.auth != null && (owner(resource.data) || confirmedFriend(resource.data.userId));
      allow create: if owner(request.resource.data)
        && request.resource.data.keys().hasAll(['userId','achievementId','unlockedAt'])
        && request.resource.data.keys().hasOnly(['userId','achievementId','unlockedAt'])
